bool MdfMemStream::Seek(int offset, int origin)
{
        switch(origin) {
                case SEEK_SET:
                {
                        if(offset < 0 || offset > nDataLen) {
                                return false;
                        }
                        nPos = offset;
                } break;
                case SEEK_CUR:
                {
                        register int test = nPos += offset;
                        if(test < 0 || test > nDataLen) {
                                return false;
                        }
                        nPos += offset;
                } break;
                case SEEK_END:
                {
                        register int test = nPos += offset;
                        if(test < 0 || test > nDataLen) {
                                return false;
                        }
                        nPos = nDataLen - origin;
                } break;
        }
        return true;
}

int MdfMemStream::Tell()
{
        return nPos;
}

int MdfMemStream::Read(void *pDest, int nLength)
{
        if(nPos + nLength >= nDataLen) {
                nLength = nDataLen - nPos;
        }
        memcpy(pDest, &pData[nPos], nLength);
        nPos += nLength;
        return nLength;
}

int MdfMemStream::Length()
{
        return nDataLen;
}

#include "MdfFormat.h"

#include <cstdio>
#include <fstream>
#include <iostream>
#include <string>

class FileStream final : public MdfStream {
public:
    explicit FileStream(const char *path) : file_(std::fopen(path, "rb")) {}
    ~FileStream() override { if (file_) std::fclose(file_); }
    bool valid() const { return file_ != nullptr; }
    bool Seek(int offset, int origin) override { return file_ && std::fseek(file_, offset, origin) == 0; }
    int Tell() override { return file_ ? static_cast<int>(std::ftell(file_)) : -1; }
    int Read(void *buffer, int size) override { return file_ ? static_cast<int>(std::fread(buffer, 1, size, file_)) : 0; }
    int Length() override {
        if (!file_) return 0;
        long current = std::ftell(file_);
        if (current < 0) return 0;
        if (std::fseek(file_, 0, SEEK_END) != 0) return 0;
        long end = std::ftell(file_);
        std::fseek(file_, current, SEEK_SET);
        return end < 0 ? 0 : static_cast<int>(end);
    }
private:
    std::FILE *file_;
};

static std::string json_escape(const std::string &value) {
    std::string result;
    for (char c : value) {
        if (c == '\\') result += "\\\\";
        else if (c == '"') result += "\\\"";
        else if (c == '\n') result += "\\n";
        else if (c == '\r') result += "\\r";
        else if (c == '\t') result += "\\t";
        else result += c;
    }
    return result;
}

int main(int argc, char **argv) {
    if (argc != 3) {
        std::cerr << "usage: mdf_converter INPUT.mdf OUTPUT.json\n";
        return 2;
    }

    FileStream input(argv[1]);
    if (!input.valid()) {
        std::cerr << "cannot open input: " << argv[1] << '\n';
        return 1;
    }
    MdfFormat model;
    if (!model.LoadFromStream(input)) {
        std::cerr << "cannot parse MDF: " << argv[1] << '\n';
        return 1;
    }

    std::ofstream output(argv[2]);
    if (!output) {
        std::cerr << "cannot open output: " << argv[2] << '\n';
        return 1;
    }
    output << "{\n"
           << "  \"format\": \"MDF\",\n"
           << "  \"format_version\": 1,\n"
           << "  \"source\": \"" << json_escape(argv[1]) << "\",\n"
