    36  bool MdfMemStream::Seek(int offset, int origin)
    37  {
    38          switch(origin) {
    39                  case SEEK_SET:
    40                  {
    41                          if(offset < 0 || offset > nDataLen) {
    42                                  return false;
    43                          }
    44                          nPos = offset;
    45                  } break;
    46                  case SEEK_CUR:
    47                  {
    48                          register int test = nPos += offset;
    49                          if(test < 0 || test > nDataLen) {
    50                                  return false;
    51                          }
    52                          nPos += offset;
    53                  } break;
    54                  case SEEK_END:
    55                  {
    56                          register int test = nPos += offset;
    57                          if(test < 0 || test > nDataLen) {
    58                                  return false;
    59                          }
    60                          nPos = nDataLen - origin;
    61                  } break;
    62          }
    63          return true;
    64  }
    65
    66  int MdfMemStream::Tell()
    67  {
    68          return nPos;
    69  }

    71  int MdfMemStream::Read(void *pDest, int nLength)
    72  {
    73          if(nPos + nLength >= nDataLen) {
    74                  nLength = nDataLen - nPos;
    75          }
    76          memcpy(pDest, &pData[nPos], nLength);
    77          nPos += nLength;
    78          return nLength;
    79  }

    82  {
    83          return nDataLen;

   376  bool MdfFormat::SaveToFile(const char *szFilename) const
   377  {
   378          FILE *fp;
   379
   380          if(m_tMdfHeader.nMeshCount == 0) {
   381                  return false;
   382          }
   383
   384          fp = fopen(szFilename, "wb");
   385          if(!fp) {
   386                  return false;
   387          }
   388
   389
   390
   391          TMdfMesh *pHeaderMeshes = (TMdfMesh*)malloc(sizeof(TMdfMesh) * m_tMdfHeader.nMeshCount);
   392          TMdfAnimation *pHeaderAnims = (TMdfAnimation*)malloc(sizeof(TMdfAnimation) * m_tMdfHeader.nAnimCount);
   393
   394          //
   395          // skip the header info for now, we'll write it later
   396          //
   397          int skipOffset = sizeof(MdfHeader);
   398          skipOffset += sizeof(TMdfMesh) * m_tMdfHeader.nMeshCount; // skip main mesh header
   399          skipOffset += sizeof(TMdfAnimation) * m_tMdfHeader.nAnimCount; // skip the animation headers
   400          skipOffset += sizeof(char *) * m_tMdfHeader.nTextureCount;  // skip the texture headers
   401          fseek(fp, skipOffset, SEEK_CUR);
   402
   403          int *pHeaderTextures = (int*)malloc(m_tMdfHeader.nTextureCount * sizeof(int));
   404          //
   405          // for each texture, save the name
   406          //
   407          for(unsigned int i = 0; i < m_tMdfHeader.nTextureCount; i++) {
   408                  size_t texLen = strlen(m_vpTextures[i]);
   409                  pHeaderTextures[i] = (int)ftell(fp);
   410                  fwrite(m_vpTextures[i], texLen+1, 1, fp);
   411          }
   412
   413          //
   414          // for each mesh, save the data
   415          //
   416          for(unsigned int i = 0; i < m_tMdfHeader.nMeshCount; i++) {
   417                  const TMdfMesh *pMesh = &m_vpMeshes[i];
   418                  pHeaderMeshes[i] = *pMesh;
   419
   420
   421                  // write out the vertices based on their type
   422                  // write the verts
   423                  ByteAlignFile(fp);
   424                  pHeaderMeshes[i].uVtxs.nDataOffset = ftell(fp);
   425
   426                  if(pMesh->meshType == TMdfMesh::MESH_SKINNED) {
   427                          fwrite(pMesh->uVtxs.pVertices, sizeof(TMdfVertSkn), pMesh->nNumVerts, fp);
   428                  } else if(pMesh->meshType == TMdfMesh::MESH_STATIC) {
   429                          fwrite(pMesh->uVtxs.pVertices, sizeof(TMdfVertSta), pMesh->nNumVerts, fp);
   430                  }
   431
   432                  // write the indices
   433                  ByteAlignFile(fp);
   434                  pHeaderMeshes[i].uIdxs.nDataOffset = ftell(fp);
   435                  fwrite(pMesh->uIdxs.pIndices, sizeof(unsigned short), pMesh->nNumIndices, fp);
   436
   437                  if(pMesh->meshType == TMdfMesh::MESH_SKINNED) {
   438                          ByteAlignFile(fp);
   439                          // write the bind-pose skin data
   440                          pHeaderMeshes[i].uBnd.nDataOffset = ftell(fp);
   441                          fwrite(pMesh->uBnd.pBindMats, sizeof(TMdfMatrix4), pMesh->nNumJoints, fp);
   442
   443                          ByteAlignFile(fp);
   444                          // write out the parenting info
   445                          pHeaderMeshes[i].uPrnt.nDataOffset = ftell(fp);
   446                          fwrite(pMesh->uPrnt.pParentInfo, sizeof(int), pMesh->nNumJoints, fp);
   447                  }
   448          }
   449
   450          //
   451          // for each animation, save the data
   452          //
   453          for(unsigned int i = 0; i < m_tMdfHeader.nAnimCount; i++) {
   454                  const TMdfAnimation *pAnim = &m_vpAnimations[i];
   455                  pHeaderAnims[i] = *pAnim;
   456
   457                  TMdfTimeline *pHeaderTimelines = (TMdfTimeline *)malloc(pAnim->nNumTimelines * sizeof(TMdfTimeline));
   458
   459                  // for each time-line
   460                  for(int t = 0; t < pAnim->nNumTimelines; t++) {
   461
   462                          const TMdfTimeline *pTimeline = &pAnim->uTls.pTimeline[t];
   463                          pHeaderTimelines[t] = *pTimeline;
   464
   465                          // write out the keyframes
   466                          ByteAlignFile(fp);
   467                          pHeaderTimelines[t].uKfs.nDataOffset = ftell(fp);
   468                          fwrite(pTimeline->uKfs.pKeyframes, sizeof(TMdfKeyframe), pTimeline->nKeyFrames, fp);
   469                  }
   470
   471                  // write out the timelines
   472                  ByteAlignFile(fp);
   473                  pHeaderAnims[i].uTls.nDataOffset = ftell(fp);
   474                  fwrite(pHeaderTimelines, sizeof(TMdfTimeline), pAnim->nNumTimelines, fp);
   475                  free(pHeaderTimelines);
   476          }
   477
   478          m_tMdfHeader.nTotalDataSize = (unsigned int)ftell(fp) - sizeof(MdfHeader);
   479
   480          //
   481          // go back and write out the headers
   482          //
   483          fseek(fp, 0, SEEK_SET);
   484
   485          // write the main header
   486          fwrite(&m_tMdfHeader, sizeof(m_tMdfHeader), 1, fp);
   487
   488          // used textures header
   489          fwrite(pHeaderTextures, sizeof(int), m_tMdfHeader.nTextureCount, fp);
   490
   491          // mesh headers
   492          fwrite(pHeaderMeshes, sizeof(TMdfMesh), m_tMdfHeader.nMeshCount, fp);
   493
   494          // animation headers
   495          fwrite(pHeaderAnims, sizeof(TMdfAnimation), m_tMdfHeader.nAnimCount, fp);
   496
   497          fclose(fp);
   498
   499          free(pHeaderMeshes);
   500          free(pHeaderAnims);
   501          free(pHeaderTextures);
   502
   503          return true;
   504  }

   506  bool MdfFormat::LoadFromStream(MdfStream &stream)
   507  {
   508          if(m_vpMeshes || m_pData) {
   509                  return false;
   510          }
   511          int nHeaderLength = sizeof(m_tMdfHeader);
   512
   513          stream.Read(&m_tMdfHeader, nHeaderLength);
   514
   515          // read the entire file
   516          m_nDataSize = stream.Length() - nHeaderLength;
   517          m_pData = (Byte*)malloc(m_nDataSize);
   518          stream.Read(m_pData, (int)m_nDataSize);
   519
   520          int dataPos = 0;
   521
   522          // check for textures and set their pointer
   523          if(m_tMdfHeader.nTextureCount > 0) {
   524                  m_vpTextures = (char**)&m_pData[dataPos];
   525                  dataPos += m_tMdfHeader.nTextureCount * sizeof(int);
   526          }
   527
   528          // check for meshes and set their pointer
   529          if(m_tMdfHeader.nMeshCount > 0) {
   530                  m_vpMeshes = (TMdfMesh*)&m_pData[dataPos];
   531                  dataPos += m_tMdfHeader.nMeshCount * sizeof(TMdfMesh);
   532          }
   533
   534          // check for animations and set their pointer
   535          if(m_tMdfHeader.nAnimCount > 0) {
   536                  m_vpAnimations = (TMdfAnimation*)&m_pData[dataPos];
   537          }
   538
   539          //
   540          // for each texture, fix up the pointers
   541          //
   542          for(size_t i = 0; i < m_tMdfHeader.nTextureCount; i++) {
   543                  m_vpTextures[i] = (char*)(m_pData  + ((int)m_vpTextures[i]) - nHeaderLength);
   544          }
   545
   546          //
   547          // for each mesh fix up the pointers
   548          //
   549          for(size_t i = 0; i < m_tMdfHeader.nMeshCount; i++) {
   550                  m_vpMeshes[i].uVtxs.pVertices = m_pData + m_vpMeshes[i].uVtxs.nDataOffset - nHeaderLength;
   551                  m_vpMeshes[i].uIdxs.pIndices = (unsigned short*)(m_pData + m_vpMeshes[i].uIdxs.nDataOffset - nHeaderLength);
   552
   553                  if(m_vpMeshes[i].nNumJoints > 0) {
   554                          m_vpMeshes[i].uBnd.pBindMats = (TMdfMatrix4*)(m_pData + m_vpMeshes[i].uBnd.nDataOffset - nHeaderLength);
   555                          m_vpMeshes[i].uPrnt.pParentInfo = (int*)(m_pData + m_vpMeshes[i].uPrnt.nDataOffset - nHeaderLength);
   556                  } else {
   557                          m_vpMeshes[i].uBnd.pBindMats = 0;
   558                          m_vpMeshes[i].uPrnt.pParentInfo = 0;
   559                  }
   560                  dataPos += sizeof(TMdfMesh);
   561          }
   562
   563          //
   564          // for each animation, fix up the pointers
   565          //
   566          for(size_t i = 0; i < m_tMdfHeader.nAnimCount; i++) {
   567                  m_vpAnimations[i].uTls.pTimeline = (TMdfTimeline*)(m_pData + m_vpAnimations[i].uTls.nDataOffset - nHeaderLength);
   568
   569                  // fix up the time-line pointers
   570                  for(int t = 0; t < m_vpAnimations[i].nNumTimelines; t++) {
   571                          m_vpAnimations[i].uTls.pTimeline[t].uKfs.pKeyframes = (TMdfKeyframe*)(m_pData + m_vpAnimations[i].uTls.pTimeline[t].uKfs.nDataOf
fset - nHeaderLength);
   572                  }
   573          }
   574
   575          return true;
   576  }

    36  bool MdfMemStream::Seek(int offset, int origin)
    37  {
    38          switch(origin) {
    39                  case SEEK_SET:
    40                  {
    41                          if(offset < 0 || offset > nDataLen) {
    42                                  return false;
    43                          }
    44                          nPos = offset;
    45                  } break;
    46                  case SEEK_CUR:
    47                  {
    48                          register int test = nPos += offset;
    49                          if(test < 0 || test > nDataLen) {
    50                                  return false;
    51                          }
    52                          nPos += offset;
    53                  } break;
    54                  case SEEK_END:
    55                  {
    56                          register int test = nPos += offset;
    57                          if(test < 0 || test > nDataLen) {
    58                                  return false;
    59                          }
    60                          nPos = nDataLen - origin;
    61                  } break;
    62          }
    63          return true;
    64  }
    65
    66  int MdfMemStream::Tell()
    67  {
    68          return nPos;
    69  }
    70
    71  int MdfMemStream::Read(void *pDest, int nLength)
    72  {
    73          if(nPos + nLength >= nDataLen) {
    74                  nLength = nDataLen - nPos;
    75          }
    76          memcpy(pDest, &pData[nPos], nLength);
    77          nPos += nLength;
    78          return nLength;
    79  }
    80
    81  int MdfMemStream::Length()
    82  {
    83          return nDataLen;
    84  }

# MDF: Model Definition Format

MDF is a custom high performance 3D mesh+animation file format library originally used in some DirectX and OpenGL
projects that I wrote back in 2006-2007, including games.

## Features

- Multiple meshes per model
- Textures (externally referenced filenames)
- Skeletal animation and animation timelines
- Efficient loading (single malloc + read for the entire file)

## Platform support

- Originally developed and tested on Win32
