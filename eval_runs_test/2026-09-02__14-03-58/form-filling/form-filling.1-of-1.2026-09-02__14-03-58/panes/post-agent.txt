from dataclasses import dataclass

FIELD_LABEL_PATTERN = re.compile(r"^([A-Za-z][A-Za-z ]{1,80}):\s*$")
WORD_PATTERN = re.compile(r"[A-Za-z0-9']+")


@dataclass(frozen=True)
class FormField:
    field_id: str
    field_name: str
    source: str


@dataclass(frozen=True)
class Candidate:
    label: str
    value: str
    source: str
    context: str


def slugify(value: str) -> str:
    slug = re.sub(r"[^a-z0-9]+", "_", value.lower()).strip("_")
    return slug or "field"


def normalize_text(value: str) -> str:
    return " ".join(value.split()).strip()


def tokenize(value: str) -> Counter:
    tokens = [token.lower() for token in WORD_PATTERN.findall(value)]
    return Counter(tokens)


def cosine_similarity(left: Counter, right: Counter) -> float:
    if not left or not right:
        return 0.0
    shared = set(left) & set(right)
    numerator = sum(left[token] * right[token] for token in shared)
    left_norm = math.sqrt(sum(weight * weight for weight in left.values()))
    right_norm = math.sqrt(sum(weight * weight for weight in right.values()))
    if left_norm == 0.0 or right_norm == 0.0:
        return 0.0
    return numerator / (left_norm * right_norm)


def analyze_pdf_fields(pdf_path: Path) -> Tuple[List[FormField], str]:
    if not pdf_path.exists():
        raise FileNotFoundError(f"PDF file not found: {pdf_path}")

    reader = PdfReader(str(pdf_path))
    fields = reader.get_fields() or {}
    if fields:
        return [
            FormField(field_id=str(field_id), field_name=str(info.get("/T") or field_id), source="acroform")
            for field_id, info in fields.items()
        ], "acroform"

    inferred: Dict[str, FormField] = {}
    for page in reader.pages:
        text = page.extract_text() or ""
        for line in text.splitlines():
            match = FIELD_LABEL_PATTERN.match(line.strip())
            if not match:
                continue
            field_name = match.group(1).strip()
            field_id = slugify(field_name)
            inferred.setdefault(
                field_id,
                FormField(field_id=field_id, field_name=field_name, source="pdf_text_label"),
            )

    if not inferred:
        raise ValueError("No fillable fields or form-like labels could be identified in the PDF")
    return list(inferred.values()), "pdf_text_label_fallback"


def load_documents(documents_dir: Path) -> Dict[str, str]:
    if not documents_dir.exists():
        raise FileNotFoundError(f"Documents directory not found: {documents_dir}")

    documents: Dict[str, str] = {}
    for path in sorted(documents_dir.glob("*.txt")):
        text = path.read_text(encoding="utf-8")
        if text.strip():
            documents[path.stem] = text
    if not documents:
        raise ValueError(f"No text documents found in {documents_dir}")
    return documents


def first_match(patterns: Sequence[str], text: str) -> Optional[str]:
    for pattern in patterns:
        match = re.search(pattern, text, flags=re.IGNORECASE | re.MULTILINE)
        if match:
            return normalize_text(match.group(1).strip(" .,-"))
    return None

        "email address": [r"(?:Email Address|Email):\s*([\w.+-]+@[\w.-]+\.[A-Za-z]{2,})"],
        "address": [r"(?:Current Address|Address):\s*(.+)$"],
        "phone": [r"(?:Phone Number|Phone \(Home\)|Phone):\s*(\(?\d{3}\)?[-\s]\d{3}[-\s]\d{4})"],
        "gender": [r"^Gender:\s*(Male|Female|Non-binary|Other)\b"],
        "date of birth": [r"Date of Birth:\s*(.+)$"],
        "nationality": [r"Nationality:\s*(.+)$"],
        "job title": [r"Job Title:\s*(.+)$", r"^Position:\s*(.+)$"],
        "employer": [r"Current Employer:\s*(.+)$", r"\*\*([^*]+Inc\.)\*\*"],
        "education": [r"###\s*(Master of Science in Computer Science)"],
        "skills": [r"Programming Languages:\s*(.+)$"],
    }

    candidates: List[Candidate] = []
    for source, text in documents.items():
        compact_context = normalize_text(text)
        for label, label_patterns in patterns.items():
            value = first_match(label_patterns, text)
            if value:
                candidates.append(
                    Candidate(
                        label=label,
                        value=value,
                        source=source,
                        context=compact_context[:600],
                    )
                )
    return candidates


def retrieve_best_candidate(
    field: FormField,
    candidates: Sequence[Candidate],
    threshold: float = 0.08,
) -> Optional[Tuple[Candidate, float]]:
    if not candidates:
        return None

    query = tokenize(f"{field.field_name} {field.field_id.replace('_', ' ')}")
    scored: List[Tuple[float, Candidate]] = []
    for candidate in candidates:
        candidate_text = f"{candidate.label} {candidate.value} {candidate.context}"
        score = cosine_similarity(query, tokenize(candidate_text))
        scored.append((score, candidate))

    scored.sort(key=lambda item: item[0], reverse=True)
    best_score, best_candidate = scored[0]
    if best_score < threshold:
        return None
    return best_candidate, best_score


def build_field_mapping(
    pdf_path: Path = PDF_PATH,
    documents_dir: Path = DOCUMENTS_DIR,
) -> Dict[str, object]:
    fields, field_detection_method = analyze_pdf_fields(pdf_path)
    documents = load_documents(documents_dir)
    candidates = extract_candidates(documents)

    mapped_fields: Dict[str, Dict[str, str]] = {}
    unmapped_fields: List[str] = []
    similarity_scores: Dict[str, float] = {}

    for field in fields:
        retrieved = retrieve_best_candidate(field, candidates)
        if retrieved is None:
            unmapped_fields.append(field.field_id)
            continue
        candidate, score = retrieved
        mapped_fields[field.field_id] = {
            "field_name": field.field_name,
            "value": candidate.value,
            "source": candidate.source,
        }
        similarity_scores[field.field_id] = round(score, 4)

    return {
        "pdf_file": pdf_path.name,
        "mapped_fields": mapped_fields,
        "unmapped_fields": unmapped_fields,
        "total_mapped": len(mapped_fields),
        "total_unmapped": len(unmapped_fields),
        "rag_stats": {
            "documents_processed": len(documents),
            "total_chars_processed": sum(len(text) for text in documents.values()),
            "retrieval_method": "regex candidate extraction + cosine similarity",
            "field_detection_method": field_detection_method,
            "candidate_count": len(candidates),
            "similarity_scores": similarity_scores,
        },
    }


def main() -> None:
    report = build_field_mapping()
    OUTPUT_PATH.write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(f"Wrote field mapping report to {OUTPUT_PATH}")


if __name__ == "__main__":
    main()
