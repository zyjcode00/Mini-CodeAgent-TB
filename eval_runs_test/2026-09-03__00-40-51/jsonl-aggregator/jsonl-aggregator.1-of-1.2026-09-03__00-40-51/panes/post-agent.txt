#!/usr/bin/env python3
"""Aggregate record JSONL files into a compact JSON summary."""

from __future__ import annotations

import argparse
import json
from collections import Counter, defaultdict
from decimal import Decimal, ROUND_HALF_UP
from pathlib import Path
from typing import Iterable


CENT = Decimal("0.01")


def aggregate_files(input_files: Iterable[Path]) -> dict:
    """Return the requested top-five user and tag aggregates."""
    user_amounts: defaultdict[str, Decimal] = defaultdict(lambda: Decimal("0"))
    user_items: defaultdict[str, int] = defaultdict(int)
    tag_counts: Counter[str] = Counter()

    for input_file in input_files:
        with input_file.open("r", encoding="utf-8") as stream:
            for line_number, line in enumerate(stream, 1):
                if not line.strip():
                    continue
                try:
                    record = json.loads(line)
                except json.JSONDecodeError as exc:
                    raise ValueError(f"Invalid JSON in {input_file}:{line_number}") from exc

                user = record["user"]
                user_amounts[user] += Decimal(str(record["amount"]))
                user_items[user] += int(record["items"])
                tag_counts.update(record.get("tags", []))

    top_users = sorted(
        user_amounts,
        key=lambda user: (-user_amounts[user], user),
    )[:5]
    top_tags = sorted(
        tag_counts,
        key=lambda tag: (-tag_counts[tag], tag),
    )[:5]

    return {
        "top_5_users_by_amount": {
            user: {
                "total_amount": float(user_amounts[user].quantize(CENT, rounding=ROUND_HALF_UP)),
                "total_items": int(user_items[user]),
            }
            for user in top_users
        },
        "top_5_tags_by_count": {
            tag: {"count": int(tag_counts[tag])}
            for tag in top_tags
        },
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--input-dir",
        type=Path,
        default=Path("/app"),
        help="Directory containing records_*.jsonl files (default: /app)",
    )
    parser.add_argument(
        "--output",
        type=Path,
        default=Path("/app/aggregates.json"),
        help="Output JSON path (default: /app/aggregates.json)",
    )
    args = parser.parse_args()

    input_files = sorted(args.input_dir.glob("records_*.jsonl"))
    if not input_files:
        raise SystemExit(f"No records_*.jsonl files found in {args.input_dir}")

    result = aggregate_files(input_files)
    args.output.write_text(
        json.dumps(result, indent=2, ensure_ascii=True) + "\n",
        encoding="utf-8",
    )


if __name__ == "__main__":
    main()
