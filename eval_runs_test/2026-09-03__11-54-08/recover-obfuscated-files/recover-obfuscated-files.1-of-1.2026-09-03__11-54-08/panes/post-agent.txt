#!/usr/bin/env python3
"""Recover files whose names and contents were encoded with Base64."""

from __future__ import annotations

import argparse
import base64
import binascii
import os
import tempfile
from pathlib import Path

SUFFIX = ".b64_content"


class RecoveryError(ValueError):
    """Raised when an obfuscated file does not have a safe valid format."""


def decode_base64(value: bytes, *, label: str, allow_whitespace: bool = False) -> bytes:
    if allow_whitespace:
        value = b"".join(value.split())
    try:
        return base64.b64decode(value, validate=True)
    except (binascii.Error, ValueError) as exc:
        raise RecoveryError(f"invalid Base64 {label}") from exc


def decode_filename(encoded: str) -> str:
    decoded = decode_base64(encoded.encode("ascii"), label="filename")
    try:
        filename = decoded.decode("utf-8")
    except UnicodeDecodeError as exc:
        raise RecoveryError("decoded filename is not UTF-8") from exc

    # Only a plain filename is allowed; decoded paths must never escape output_dir.
    if not filename or filename in {".", ".."} or "/" in filename or "\\" in filename or "\x00" in filename:
        raise RecoveryError(f"unsafe decoded filename: {filename!r}")
    return filename


def recover_file(source: Path, output_dir: Path) -> Path:
    if not source.name.endswith(SUFFIX):
        raise RecoveryError(f"unsupported source file: {source.name}")
    encoded_name = source.name[: -len(SUFFIX)]
    filename = decode_filename(encoded_name)
    content = decode_base64(source.read_bytes(), label=f"content of {source.name}", allow_whitespace=True)

    output_dir.mkdir(parents=True, exist_ok=True)
    destination = output_dir / filename
    # Replace only a regular output path; a symlink could redirect writes elsewhere.
    if destination.exists() and destination.is_symlink():
        raise RecoveryError(f"refusing to overwrite symlink: {destination}")
    with tempfile.NamedTemporaryFile(dir=output_dir, prefix=f".{filename}.", delete=False) as handle:
        temporary = Path(handle.name)
        handle.write(content)
        handle.flush()
        os.fsync(handle.fileno())
    try:
        os.replace(temporary, destination)
    except Exception:
        temporary.unlink(missing_ok=True)
        raise
    return destination


def recover_directory(input_dir: Path, output_dir: Path) -> list[Path]:
    if not input_dir.is_dir():
        raise RecoveryError(f"input directory does not exist: {input_dir}")
    sources = sorted(path for path in input_dir.iterdir() if path.is_file() and path.name.endswith(SUFFIX))
    recovered = []
    for source in sources:
        recovered.append(recover_file(source, output_dir))
    return recovered


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("input_dir", nargs="?", type=Path, default=Path("/app/sensitive_data"))
    parser.add_argument("output_dir", nargs="?", type=Path, default=Path("/app/recovered"))
    args = parser.parse_args()
    try:
        recovered = recover_directory(args.input_dir, args.output_dir)
    except (OSError, RecoveryError) as exc:
        parser.error(str(exc))
    for path in recovered:
        print(path)
    print(f"Recovered {len(recovered)} file(s).")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
