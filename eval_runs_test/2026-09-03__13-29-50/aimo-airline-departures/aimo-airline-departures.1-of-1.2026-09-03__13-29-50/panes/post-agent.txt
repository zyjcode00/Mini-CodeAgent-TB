import importlib.util
from pathlib import Path


MODULE_PATH = Path('/app/airline_analysis.py')


def load_module():
    spec = importlib.util.spec_from_file_location('airline_analysis', MODULE_PATH)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def test_longest_no_flight_streak_for_aligned_departures():
    module = load_module()
    assert module.longest_no_flight_streak((0, 0, 0)) == 99


def test_solver_finds_guaranteed_maximum():
    module = load_module()
    assert module.solve() == 79


def test_main_writes_only_final_number(tmp_path):
    module = load_module()
    original_path = module.RESULT_PATH
    try:
        module.RESULT_PATH = tmp_path / 'results.txt'
        module.main()
        assert module.RESULT_PATH.read_text(encoding='utf-8') == '83\n'
    finally:
        module.RESULT_PATH = original_path
