import time

import mlflow
import mlflow.sklearn
import numpy as np
from mlflow.models import infer_signature
from mlflow.tracking import MlflowClient
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
from sklearn.model_selection import train_test_split


TRACKING_URI = "http://127.0.0.1:8080"
REGISTERED_MODEL_NAME = "gpt-5"
EXPERIMENT_NAME = "gpt-5-random-linear-regression"
RANDOM_SEED = None  # None intentionally randomizes each run.


def main() -> None:
    rng = np.random.default_rng(RANDOM_SEED)

    # Random dataset with exactly 3 features.
    n_samples = 200
    n_features = 3
    x = rng.normal(size=(n_samples, n_features))

    # Random underlying linear relationship plus random noise.
    true_coef = rng.normal(size=n_features)
    true_intercept = float(rng.normal())
    noise = rng.normal(scale=0.15, size=n_samples)
    y = x @ true_coef + true_intercept + noise

    x_train, x_test, y_train, y_test = train_test_split(
        x, y, test_size=0.25, random_state=int(rng.integers(0, 2**31 - 1))
    )

    # sklearn LinearRegression itself has no random_state parameter; randomness here comes
    # from the random training data and random train/test split above.
    model = LinearRegression()
    model.fit(x_train, y_train)

    preds = model.predict(x_test)
    mse = mean_squared_error(y_test, preds)
    r2 = r2_score(y_test, preds)

    mlflow.set_tracking_uri(TRACKING_URI)
    mlflow.set_experiment(EXPERIMENT_NAME)

    signature = infer_signature(x_train, model.predict(x_train))
    input_example = x_train[:5]

    with mlflow.start_run(run_name="random-3-feature-linear-regression") as run:
        mlflow.log_param("n_samples", n_samples)
        mlflow.log_param("n_features", n_features)
        mlflow.log_param("model_type", "sklearn.linear_model.LinearRegression")
        mlflow.log_param("true_coef", true_coef.tolist())
        mlflow.log_param("true_intercept", true_intercept)
        mlflow.log_metric("mse", float(mse))
        mlflow.log_metric("r2", float(r2))

        mlflow.sklearn.log_model(
            sk_model=model,
            artifact_path="model",
            signature=signature,
            input_example=input_example,
            registered_model_name=REGISTERED_MODEL_NAME,
        )

        print(f"run_id={run.info.run_id}")
        print(f"registered_model_name={REGISTERED_MODEL_NAME}")
        print(f"coef={model.coef_.tolist()}")
        print(f"intercept={float(model.intercept_)}")
        print(f"mse={float(mse)}")
        print(f"r2={float(r2)}")

    client = MlflowClient(tracking_uri=TRACKING_URI)
    deadline = time.time() + 60
    while time.time() < deadline:
        versions = client.search_model_versions(f"name = '{REGISTERED_MODEL_NAME}'")
        if versions:
            newest = max(versions, key=lambda v: int(v.version))
            print(f"latest_model_version={newest.version}")
            print(f"latest_model_status={newest.status}")
            return
        time.sleep(1)

    raise RuntimeError(f"Model {REGISTERED_MODEL_NAME!r} was not registered within timeout")


if __name__ == "__main__":
    main()
