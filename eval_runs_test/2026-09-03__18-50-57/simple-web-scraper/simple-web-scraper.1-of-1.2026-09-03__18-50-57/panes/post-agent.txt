#!/usr/bin/env python3
"""Scrape book information from the local bookstore website.

The scraper fetches http://server:8000, extracts book title, author, year,
rating, and price from the documented HTML structure, then writes:
  - /app/books.csv
  - /app/report.txt
"""

from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path
from typing import Iterable, Optional

import pandas as pd
import requests
from bs4 import BeautifulSoup

URL = "http://server:8000"
CSV_PATH = Path("/app/books.csv")
REPORT_PATH = Path("/app/report.txt")
CSV_COLUMNS = ["title", "author", "year", "rating", "price"]


@dataclass(frozen=True)
class Book:
    title: str
    author: str
    year: int
    rating: float
    price: float


def _required_text(parent, selector: str) -> str:
    """Return stripped text for a required selector or raise ValueError."""
    element = parent.select_one(selector)
    if element is None:
        raise ValueError(f"Missing required element: {selector}")
    return element.get_text(strip=True)


def _first_number(text: str, *, as_float: bool = False):
    """Extract the first integer/decimal number from text."""
    match = re.search(r"\d+(?:\.\d+)?", text)
    if not match:
        raise ValueError(f"No numeric value found in: {text!r}")
    value = match.group(0)
    return float(value) if as_float else int(float(value))


def clean_author(text: str) -> str:
    """Normalize author text such as 'by Jane Austen' to 'Jane Austen'."""
    return re.sub(r"^\s*by\s+", "", text, flags=re.IGNORECASE).strip()


def parse_book(book_element) -> Book:
    """Parse a single <div class='book'> element into a Book."""
    title = _required_text(book_element, ".book-title")
    author = clean_author(_required_text(book_element, ".book-author"))
    year = _first_number(_required_text(book_element, ".book-year"))
    rating = _first_number(_required_text(book_element, ".book-rating"), as_float=True)
    price = _first_number(_required_text(book_element, ".book-price"), as_float=True)
    return Book(title=title, author=author, year=year, rating=rating, price=price)


def parse_books(html: str) -> list[Book]:
    """Parse all books from the bookstore HTML."""
    soup = BeautifulSoup(html, "html.parser")
    return [parse_book(book_div) for book_div in soup.select("div.book")]


def fetch_html(url: str = URL, timeout: int = 10) -> str:
    """Fetch bookstore HTML and raise for HTTP/network failures."""
    response = requests.get(url, timeout=timeout)
    response.raise_for_status()
    return response.text


def books_to_dataframe(books: Iterable[Book]) -> pd.DataFrame:
    """Convert parsed books to a DataFrame with the required CSV columns."""
    rows = [book.__dict__ for book in books]
    return pd.DataFrame(rows, columns=CSV_COLUMNS)


def write_books_csv(books: Iterable[Book], csv_path: Path = CSV_PATH) -> pd.DataFrame:
    """Write books to CSV and return the DataFrame that was written."""
    df = books_to_dataframe(books)
    csv_path.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(csv_path, index=False)
    return df


def highest_rated_book(df: pd.DataFrame) -> Optional[pd.Series]:
    """Return the row for the highest rated book, or None when empty."""
    if df.empty:
        return None
    return df.loc[df["rating"].idxmax()]


def build_report(df: pd.DataFrame) -> str:
    """Build the requested summary report text."""
    total_books = len(df)
    average_price = 0.0 if df.empty else float(df["price"].mean())
    highest = highest_rated_book(df)

    lines = [
        f"Total number of books scraped: {total_books}",
        f"Average book price: ${average_price:.2f}",
    ]

    if highest is None:
        lines.append("Highest rated book: N/A")
    else:
        lines.append(
            "Highest rated book: "
            f"{highest['title']} by {highest['author']} "
            f"(Rating: {float(highest['rating']):.1f}/5)"
        )

    return "\n".join(lines) + "\n"


def write_report(df: pd.DataFrame, report_path: Path = REPORT_PATH) -> str:
    """Write the summary report and return its content."""
    report = build_report(df)
    report_path.parent.mkdir(parents=True, exist_ok=True)
    report_path.write_text(report, encoding="utf-8")
    return report


def scrape(url: str = URL, csv_path: Path = CSV_PATH, report_path: Path = REPORT_PATH) -> pd.DataFrame:
    """Fetch, parse, save CSV, and save report."""
    html = fetch_html(url)
    books = parse_books(html)
    df = write_books_csv(books, csv_path)
    write_report(df, report_path)
    return df


def main() -> None:
    df = scrape()
    print(f"Scraped {len(df)} books")
    print(f"Wrote {CSV_PATH}")
    print(f"Wrote {REPORT_PATH}")


if __name__ == "__main__":
    main()
