#!/usr/bin/env python3
"""Merge user data from JSON, CSV, and Parquet sources.

Outputs:
- /app/merged_users.parquet
- /app/conflicts.json
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import pandas as pd

SOURCE_CONFIG = [
    {
        "name": "source_a",
        "path": Path("/data/source_a/users.json"),
        "format": "json",
        "mapping": {
            "id": "user_id",
            "full_name": "name",
            "email": "email",
            "registration_date": "created_date",
            "status": "status",
        },
    },
    {
        "name": "source_b",
        "path": Path("/data/source_b/users.csv"),
        "format": "csv",
        "mapping": {
            "user_id": "user_id",
            "name": "name",
            "email_address": "email",
            "created_at": "created_date",
        },
    },
    {
        "name": "source_c",
        "path": Path("/data/source_c/users.parquet"),
        "format": "parquet",
        "mapping": {
            "userId": "user_id",
            "userName": "name",
            "email": "email",
            "joined": "created_date",
        },
    },
]

OUTPUT_COLUMNS = ["user_id", "name", "email", "created_date", "status"]
MERGED_OUTPUT = Path("/app/merged_users.parquet")
CONFLICT_OUTPUT = Path("/app/conflicts.json")


def normalize_scalar(value: Any) -> Any:
    """Convert pandas/numpy missing values to None and scalars to Python values."""
    if value is None:
        return None
    try:
        if pd.isna(value):
            return None
    except (TypeError, ValueError):
        pass
    if hasattr(value, "item"):
        try:
            return value.item()
        except Exception:
            pass
    return value


def normalize_date(value: Any) -> str | None:
    """Normalize date-like values to YYYY-MM-DD strings."""
    value = normalize_scalar(value)
    if value in (None, ""):
        return None
    parsed = pd.to_datetime(value, errors="coerce")
    if pd.isna(parsed):
        return str(value)
    return parsed.strftime("%Y-%m-%d")


def read_source(config: dict[str, Any]) -> pd.DataFrame:
    """Read one source and rename fields to the unified schema."""
    path = config["path"]
    if config["format"] == "json":
        df = pd.read_json(path)
    elif config["format"] == "csv":
        df = pd.read_csv(path)
    elif config["format"] == "parquet":
        df = pd.read_parquet(path)
    else:
        raise ValueError(f"Unsupported format: {config['format']}")

    df = df.rename(columns=config["mapping"])
    keep_cols = [col for col in OUTPUT_COLUMNS if col in df.columns]
    df = df[keep_cols].copy()

    if "user_id" not in df.columns:
        raise ValueError(f"Source {config['name']} has no user id column after mapping")

    df["user_id"] = df["user_id"].map(normalize_scalar).astype("int64")
    if "created_date" in df.columns:
        df["created_date"] = df["created_date"].map(normalize_date)
    for col in ["name", "email", "status"]:
        if col in df.columns:
            df[col] = df[col].map(lambda x: None if normalize_scalar(x) is None else str(normalize_scalar(x)))

    df["_source"] = config["name"]
    return df


def build_source_records(source_frames: list[pd.DataFrame]) -> dict[int, dict[str, dict[str, Any]]]:
    """Build nested records: user_id -> source -> field values."""
    records: dict[int, dict[str, dict[str, Any]]] = {}
    for df in source_frames:
        for _, row in df.iterrows():
            user_id = int(row["user_id"])
            source_name = str(row["_source"])
            values: dict[str, Any] = {}
            for col in OUTPUT_COLUMNS:
                if col in row.index:
                    value = normalize_scalar(row[col])
                    if value is not None:
                        values[col] = int(value) if col == "user_id" else value
            records.setdefault(user_id, {})[source_name] = values
    return records


def select_by_priority(user_sources: dict[str, dict[str, Any]], field: str) -> Any:
    """Select the first non-null field value according to source priority."""
    for config in SOURCE_CONFIG:
        source_name = config["name"]
        if source_name in user_sources and field in user_sources[source_name]:
            return user_sources[source_name][field]
    return None


def detect_conflicts(records: dict[int, dict[str, dict[str, Any]]]) -> list[dict[str, Any]]:
    """Detect fields whose non-null values differ across sources for the same user."""
    conflicts: list[dict[str, Any]] = []
    for user_id in sorted(records):
        user_sources = records[user_id]
        if len(user_sources) < 2:
            continue
        for field in ["name", "email", "created_date", "status"]:
            values = {
                source_name: source_values[field]
                for source_name, source_values in user_sources.items()
                if field in source_values and source_values[field] is not None
            }
            distinct_values = {str(value) for value in values.values()}
            if len(distinct_values) > 1:
                conflicts.append(
                    {
                        "user_id": user_id,
                        "field": field,
                        "values": values,
                        "selected": select_by_priority(user_sources, field),
                    }
                )
    return conflicts


def merge_users() -> tuple[pd.DataFrame, dict[str, Any]]:
    """Merge all sources and write Parquet and conflict report outputs."""
    source_frames = [read_source(config) for config in SOURCE_CONFIG]
    records = build_source_records(source_frames)

    merged_rows: list[dict[str, Any]] = []
    for user_id in sorted(records):
        user_sources = records[user_id]
        row = {field: select_by_priority(user_sources, field) for field in OUTPUT_COLUMNS}
        row["user_id"] = int(user_id)
        merged_rows.append(row)

    merged_df = pd.DataFrame(merged_rows, columns=OUTPUT_COLUMNS)
    merged_df["user_id"] = merged_df["user_id"].astype("int64")
    for col in ["name", "email", "created_date", "status"]:
        merged_df[col] = merged_df[col].astype("string")

    conflicts = detect_conflicts(records)
    conflict_report = {
        "total_conflicts": len(conflicts),
        "conflicts": conflicts,
    }

    MERGED_OUTPUT.parent.mkdir(parents=True, exist_ok=True)
    merged_df.to_parquet(MERGED_OUTPUT, index=False)
    with CONFLICT_OUTPUT.open("w", encoding="utf-8") as f:
        json.dump(conflict_report, f, indent=2, ensure_ascii=False)
        f.write("\n")

    return merged_df, conflict_report


if __name__ == "__main__":
    df, report = merge_users()
    print(f"Wrote {MERGED_OUTPUT} with {len(df)} rows")
    print(f"Wrote {CONFLICT_OUTPUT} with {report['total_conflicts']} conflicts")

import json
import re
from pathlib import Path

import pandas as pd

from merge_users import CONFLICT_OUTPUT, MERGED_OUTPUT, merge_users, read_source, SOURCE_CONFIG


def setup_module(module):
    # Regenerate outputs so tests validate the current implementation and files.
    merge_users()


def test_output_files_exist_and_have_expected_schema():
    assert MERGED_OUTPUT.exists(), f"Missing {MERGED_OUTPUT}"
    assert CONFLICT_OUTPUT.exists(), f"Missing {CONFLICT_OUTPUT}"

    df = pd.read_parquet(MERGED_OUTPUT)
    assert list(df.columns) == ["user_id", "name", "email", "created_date", "status"]
    assert pd.api.types.is_integer_dtype(df["user_id"]), df.dtypes


def test_all_unique_users_are_included_once():
    source_ids = set()
    for config in SOURCE_CONFIG:
        normalized = read_source(config)
        source_ids.update(normalized["user_id"].astype(int).tolist())

    df = pd.read_parquet(MERGED_OUTPUT)
    output_ids = df["user_id"].astype(int).tolist()

    assert set(output_ids) == source_ids
    assert len(output_ids) == len(set(output_ids))
    assert len(output_ids) == 4


def test_field_mappings_and_priority_resolution():
    df = pd.read_parquet(MERGED_OUTPUT).set_index("user_id")

    # user 101 appears in all sources. All selected values should come from source_a.
    assert df.loc[101, "name"] == "John Doe"
    assert df.loc[101, "email"] == "john@a.com"
    assert df.loc[101, "created_date"] == "2024-01-15"
    assert df.loc[101, "status"] == "active"

    # source_b-only user validates user_id/name/email_address/created_at mapping.
    assert df.loc[103, "name"] == "Bob Wilson"
    assert df.loc[103, "email"] == "bob@b.com"
    assert df.loc[103, "created_date"] == "2024-03-01"

    # source_c-only user validates userId/userName/email/joined mapping.
    assert df.loc[104, "name"] == "Alice Brown"
    assert df.loc[104, "email"] == "alice@c.com"
    assert df.loc[104, "created_date"] == "2024-04-01"


def test_created_date_format_is_yyyy_mm_dd():
    df = pd.read_parquet(MERGED_OUTPUT)
    pattern = re.compile(r"^\d{4}-\d{2}-\d{2}$")
    assert df["created_date"].notna().all()
    assert df["created_date"].map(lambda value: bool(pattern.match(str(value)))).all()


def test_conflict_report_structure_count_and_selected_values():
    with open(CONFLICT_OUTPUT, encoding="utf-8") as f:
        report = json.load(f)

    assert set(report) == {"total_conflicts", "conflicts"}
    assert report["total_conflicts"] == len(report["conflicts"])
    assert report["total_conflicts"] == 3

    conflicts_by_field = {(item["user_id"], item["field"]): item for item in report["conflicts"]}
    assert set(conflicts_by_field) == {
        (101, "name"),
        (101, "email"),
        (101, "created_date"),
    }

    assert conflicts_by_field[(101, "name")] == {
        "user_id": 101,
        "field": "name",
        "values": {
            "source_a": "John Doe",
            "source_b": "John Doe",
            "source_c": "John D.",
        },
        "selected": "John Doe",
    }
    assert conflicts_by_field[(101, "email")]["selected"] == "john@a.com"
    assert conflicts_by_field[(101, "created_date")]["selected"] == "2024-01-15"


def test_conflict_report_values_are_actual_differing_source_values():
    with open(CONFLICT_OUTPUT, encoding="utf-8") as f:
        report = json.load(f)

    for conflict in report["conflicts"]:
        values = conflict["values"]
        assert len(values) >= 2
        assert len({str(value) for value in values.values()}) > 1
        # The selected value must be present in the highest-priority available source.
        for source_name in ["source_a", "source_b", "source_c"]:
            if source_name in values:
                assert conflict["selected"] == values[source_name]
                break
