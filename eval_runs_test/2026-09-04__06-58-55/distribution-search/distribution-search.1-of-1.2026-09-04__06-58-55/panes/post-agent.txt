#!/usr/bin/env python3
"""Generate a probability distribution with target KL divergences.

We construct a 150,000-dimensional distribution with three probability levels:
  - one high-probability token: a
  - m medium-probability tokens: b
  - the remaining tokens: c

For fixed m, (a, b, c) are determined by three equations:
  1. total probability is 1
  2. KL(P || U) = 10
  3. KL(U || P) = 10

The root solve is done in transformed variables to maintain positivity and the
normalization equation is enforced by construction.
"""

from __future__ import annotations

import numpy as np
from scipy.optimize import root

N = 150_000
TARGET = 10.0
OUT = "/app/dist.npy"


def kl_forward(p: np.ndarray) -> float:
    """KL(P || U) for uniform U over len(p) items."""
    n = p.size
    return float(np.sum(p * np.log(p * n)))


def kl_backward(p: np.ndarray) -> float:
    """KL(U || P) for uniform U over len(p) items."""
    n = p.size
    return float(np.mean(np.log((1.0 / n) / p)))


def values_from_unconstrained(x: np.ndarray, m: int) -> tuple[float, float, float]:
    """Map unconstrained x=(u,v) to positive a,b,c summing to 1.

    Let r=exp(u)=b/a and s=exp(v)=c/a. Then
        a + m*b + rest*c = a * (1 + m*r + rest*s) = 1.
    """
    rest = N - 1 - m
    u, v = x
    r = np.exp(u)
    s = np.exp(v)
    denom = 1.0 + m * r + rest * s
    a = 1.0 / denom
    b = r / denom
    c = s / denom
    return float(a), float(b), float(c)


def equations(x: np.ndarray, m: int) -> np.ndarray:
    a, b, c = values_from_unconstrained(x, m)
    rest = N - 1 - m
    fwd = a * np.log(a * N) + m * b * np.log(b * N) + rest * c * np.log(c * N)
    bwd = (np.log(1.0 / (N * a)) + m * np.log(1.0 / (N * b)) + rest * np.log(1.0 / (N * c))) / N
    return np.array([fwd - TARGET, bwd - TARGET], dtype=np.float64)


def solve_levels() -> tuple[int, float, float, float]:
    """Find m and levels a,b,c satisfying both KL targets."""
    # Several initial guesses make the script robust/reproducible.  The final
    # solution currently lands at m=5 with one large, five medium, and the rest
    # tiny probabilities.
    guesses = [
        (5, np.array([-1.5, -24.0])),
        (5, np.array([-2.0, -25.0])),
        (6, np.array([-1.8, -24.0])),
        (4, np.array([-1.2, -24.5])),
        (7, np.array([-2.2, -24.0])),
        (10, np.array([-2.5, -24.0])),
    ]
    best = None
    for m, x0 in guesses:
        sol = root(lambda x: equations(x, m), x0, method="hybr", tol=1e-12)
        residual = equations(sol.x, m)
        err = float(np.max(np.abs(residual)))
        if best is None or err < best[0]:
            best = (err, m, sol)
        if sol.success and err < 1e-10:
            a, b, c = values_from_unconstrained(sol.x, m)
            return m, a, b, c

    assert best is not None
    err, m, sol = best
    raise RuntimeError(f"Could not solve KL equations; best m={m}, err={err}, message={sol.message}")


def build_distribution() -> np.ndarray:
    m, a, b, c = solve_levels()
    rest = N - 1 - m
    p = np.empty(N, dtype=np.float64)
    p[0] = a
    p[1 : 1 + m] = b
    p[1 + m :] = c

    # Correct any last-bit summation drift without changing positivity.
    p[-1] += 1.0 - float(p.sum())

    fwd = kl_forward(p)
    bwd = kl_backward(p)
    if p.shape != (N,) or not np.all(np.isfinite(p)) or np.any(p <= 0):
        raise RuntimeError("Generated distribution is not strictly positive and finite")
    if abs(float(p.sum()) - 1.0) > 1e-12:
        raise RuntimeError(f"Generated distribution is not normalized: sum={p.sum()!r}")
    if abs(fwd - TARGET) > 1e-6 or abs(bwd - TARGET) > 1e-6:
        raise RuntimeError(f"KL targets not met tightly: forward={fwd}, backward={bwd}")

    print(f"m={m}")
    print(f"levels: a={a:.17g}, b={b:.17g}, c={c:.17g}")
    print(f"sum={p.sum():.17g}, min={p.min():.17g}, max={p.max():.17g}")
    print(f"forward_kl={fwd:.17g}")
    print(f"backward_kl={bwd:.17g}")
    return p


def main() -> None:
    p = build_distribution()
    np.save(OUT, p)
    print(f"saved {OUT}")


if __name__ == "__main__":
    main()
