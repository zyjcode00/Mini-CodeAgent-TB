#!/usr/bin/env python3
"""Merge heterogeneous user data sources with priority-based conflict handling."""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, Iterable, List, Tuple

def is_missing(value: Any) -> bool:
    """Return True for null/NaN/empty-string values that should not override data."""
    if value is None:
        return True
    try:
        if pd.isna(value):
            return True
    except (TypeError, ValueError):
        pass
    if isinstance(value, str) and value.strip() == "":
        return True
    return False


def normalize_scalar(field: str, value: Any) -> Any:
    """Normalize a scalar value for stable merging and reporting."""
    if is_missing(value):
        return None
    if field == "user_id":
        return int(value)
    if field == "created_date":
        return pd.to_datetime(value).strftime("%Y-%m-%d")
    if isinstance(value, str):
        return value.strip()
    # Convert numpy scalar values to native Python types for JSON serialization.
    if hasattr(value, "item"):
        return value.item()
    return value


def read_source(path: Path, source_format: str) -> pd.DataFrame:
    """Read a source file into a pandas DataFrame."""
    if source_format == "json":
        with path.open("r", encoding="utf-8") as fh:
            data = json.load(fh)
        if isinstance(data, dict):
            # Support either a bare object keyed by users or a wrapper like {"users": [...]}.
            for candidate in ("users", "data", "records"):
                if isinstance(data.get(candidate), list):
                    data = data[candidate]
                    break
        return pd.DataFrame(data)
    if source_format == "csv":
        return pd.read_csv(path)
    if source_format == "parquet":
        return pd.read_parquet(path)
    raise ValueError(f"Unsupported source format: {source_format}")


def normalize_frame(df: pd.DataFrame) -> List[Dict[str, Any]]:
    """Map source-specific columns to the canonical output schema."""
    rows: List[Dict[str, Any]] = []
    for raw_record in df.to_dict(orient="records"):
        normalized: Dict[str, Any] = {}
        for raw_field, raw_value in raw_record.items():
            canonical_field = FIELD_MAP.get(raw_field)
            if canonical_field is None:
                continue
            normalized[canonical_field] = normalize_scalar(canonical_field, raw_value)
        if "user_id" not in normalized or normalized["user_id"] is None:
            continue
        rows.append(normalized)
    return rows


def source_records() -> Dict[str, Dict[int, Dict[str, Any]]]:
    """Return records indexed by source name then user_id."""
    indexed: Dict[str, Dict[int, Dict[str, Any]]] = {}
    for source_name, path, source_format in SOURCE_CONFIG:
        df = read_source(path, source_format)
        records = normalize_frame(df)
        by_id: Dict[int, Dict[str, Any]] = {}
        for record in records:
            user_id = int(record["user_id"])
            # If a source unexpectedly contains duplicates, keep first non-null values.
            if user_id not in by_id:
                by_id[user_id] = {"user_id": user_id}
            for field, value in record.items():
                if field == "user_id":
                    continue
                if not is_missing(value) and is_missing(by_id[user_id].get(field)):
                    by_id[user_id][field] = value
        indexed[source_name] = by_id
    return indexed


def selected_value(values_by_source: Dict[str, Any]) -> Tuple[Any, str | None]:
    """Select the highest-priority non-missing value."""
    for source_name, _, _ in SOURCE_CONFIG:
        value = values_by_source.get(source_name)
        if not is_missing(value):
            return value, source_name
    return None, None


def merge_records(records_by_source: Dict[str, Dict[int, Dict[str, Any]]]) -> Tuple[pd.DataFrame, Dict[str, Any]]:
    """Merge records and build the conflict report."""
    all_user_ids = sorted({uid for source_records_ in records_by_source.values() for uid in source_records_})
    merged_rows: List[Dict[str, Any]] = []
    conflicts: List[Dict[str, Any]] = []

    for user_id in all_user_ids:
        merged: Dict[str, Any] = {"user_id": int(user_id)}
        for field in MERGE_FIELDS:
            values_by_source = {
                source_name: records_by_source.get(source_name, {}).get(user_id, {}).get(field)
                for source_name, _, _ in SOURCE_CONFIG
                if not is_missing(records_by_source.get(source_name, {}).get(user_id, {}).get(field))
            }
            selected, _ = selected_value(values_by_source)
            merged[field] = selected

            distinct_values = []
            for value in values_by_source.values():
                if value not in distinct_values:
                    distinct_values.append(value)
            if len(distinct_values) > 1:
                conflicts.append(
                    {
                        "user_id": int(user_id),
                        "field": field,
                        "values": values_by_source,
                        "selected": selected,
                    }
                )
        merged_rows.append(merged)

    merged_df = pd.DataFrame(merged_rows, columns=OUTPUT_COLUMNS)
    merged_df["user_id"] = merged_df["user_id"].astype("int64")
    for column in ["name", "email", "created_date", "status"]:
        merged_df[column] = merged_df[column].astype("string")

    conflict_report = {"total_conflicts": len(conflicts), "conflicts": conflicts}
    return merged_df, conflict_report
