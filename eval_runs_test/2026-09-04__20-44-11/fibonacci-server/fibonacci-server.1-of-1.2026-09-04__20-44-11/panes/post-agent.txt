#!/usr/bin/env python3
"""Simple Fibonacci HTTP server.

Runs on port 3000 by default and exposes a single GET endpoint:
    /fib?n={non-negative integer}
"""

from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
from urllib.parse import parse_qs, urlparse

HOST = "0.0.0.0"
PORT = 3000


def fibonacci(n: int) -> int:
    """Return the nth Fibonacci number, using F(0)=0 and F(1)=1."""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a


def parse_non_negative_integer(raw_value: str) -> int:
    """Parse a query string value as a non-negative integer.

    Raises ValueError if the value is missing, non-integral, or negative.
    """
    if raw_value is None or raw_value == "":
        raise ValueError("n query parameter is required")

    if not raw_value.isdecimal():
        raise ValueError("n query parameter must be a non-negative integer")

    return int(raw_value)


class FibRequestHandler(BaseHTTPRequestHandler):
    """HTTP handler exposing GET /fib."""

    def do_GET(self) -> None:
        parsed_url = urlparse(self.path)

        if parsed_url.path != "/fib":
            self.send_json({"error": "Not Found"}, status=404)
            return

        query = parse_qs(parsed_url.query, keep_blank_values=True)
        raw_n = query.get("n", [None])[0]

        try:
            n = parse_non_negative_integer(raw_n)
        except ValueError as exc:
            self.send_json({"error": str(exc)}, status=400)
            return

        self.send_json({"result": fibonacci(n)})

    def send_json(self, payload: dict, status: int = 200) -> None:
        body = json.dumps(payload).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, format: str, *args) -> None:
        """Keep test/server output concise by suppressing default access logs."""
        return


def run_server(host: str = HOST, port: int = PORT) -> None:
    server = ThreadingHTTPServer((host, port), FibRequestHandler)
    print(f"Serving Fibonacci API on http://{host}:{port}", flush=True)
    server.serve_forever()


if __name__ == "__main__":
    run_server()
