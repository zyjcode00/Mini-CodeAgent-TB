#!/usr/bin/env python3
"""Load and print details for the internal development TLS certificate."""

from __future__ import annotations

import ssl
from datetime import datetime, timezone
from pathlib import Path

CERT_PATH = Path("/app/ssl/server.crt")
EXPECTED_CN = "dev-internal.company.local"


def _format_as_yyyy_mm_dd(openssl_time: str) -> str:
    """Convert an OpenSSL ASN.1 time string into YYYY-MM-DD format."""
    parsed = datetime.strptime(openssl_time, "%b %d %H:%M:%S %Y %Z")
    return parsed.replace(tzinfo=timezone.utc).strftime("%Y-%m-%d")


def _get_subject_value(cert_info: dict, field_name: str) -> str | None:
    """Return a value from ssl._test_decode_cert subject tuples."""
    for relative_distinguished_name in cert_info.get("subject", ()):  # tuple of tuples
        for key, value in relative_distinguished_name:
            if key == field_name:
                return value
    return None


def main() -> int:
    if not CERT_PATH.exists():
        raise FileNotFoundError(f"Certificate not found: {CERT_PATH}")

    cert_info = ssl._ssl._test_decode_cert(str(CERT_PATH))  # type: ignore[attr-defined]
    common_name = _get_subject_value(cert_info, "commonName")
    organization = _get_subject_value(cert_info, "organizationName")
    not_before = _format_as_yyyy_mm_dd(cert_info["notBefore"])
    not_after = _format_as_yyyy_mm_dd(cert_info["notAfter"])

    if common_name != EXPECTED_CN:
        raise ValueError(f"Unexpected Common Name: {common_name!r}")

    print(f"Certificate path: {CERT_PATH}")
    print(f"Organization: {organization}")
    print(f"Common Name: {common_name}")
    print(f"Valid From: {not_before}")
    print(f"Expiration Date: {not_after}")
    print("Certificate verification successful")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

#!/usr/bin/env python3
"""Requirement checks for the self-signed TLS certificate task."""

from __future__ import annotations

import os
import re
import ssl
import stat
import subprocess
from datetime import datetime, timezone
from pathlib import Path

SSL_DIR = Path("/app/ssl")
KEY = SSL_DIR / "server.key"
CRT = SSL_DIR / "server.crt"
PEM = SSL_DIR / "server.pem"
VERIFICATION = SSL_DIR / "verification.txt"
CHECK_SCRIPT = Path("/app/check_cert.py")
EXPECTED_CN = "dev-internal.company.local"
EXPECTED_ORG = "DevOps Team"


def run(command: list[str]) -> str:
    result = subprocess.run(command, check=True, text=True, capture_output=True)
    return result.stdout.strip()


def parse_openssl_date(value: str) -> datetime:
    return datetime.strptime(value, "%b %d %H:%M:%S %Y %Z").replace(tzinfo=timezone.utc)


def test_required_files_exist() -> None:
    assert SSL_DIR.is_dir(), "/app/ssl directory is missing"
    for path in (KEY, CRT, PEM, VERIFICATION, CHECK_SCRIPT):
        assert path.is_file(), f"missing required file: {path}"


def test_private_key_is_2048_bit_rsa_and_mode_600() -> None:
    mode = stat.S_IMODE(KEY.stat().st_mode)
    assert mode == 0o600, f"server.key permissions should be 600, got {oct(mode)}"
    key_text = run(["openssl", "rsa", "-in", str(KEY), "-text", "-noout"])
    assert "Private-Key: (2048 bit" in key_text
    assert run(["openssl", "rsa", "-in", str(KEY), "-check", "-noout"]) == "RSA key ok"


def test_certificate_subject_and_validity() -> None:
    subject = run(["openssl", "x509", "-in", str(CRT), "-noout", "-subject"])
    assert f"O = {EXPECTED_ORG}" in subject
    assert f"CN = {EXPECTED_CN}" in subject

    dates = run(["openssl", "x509", "-in", str(CRT), "-noout", "-dates"])
    fields = dict(line.split("=", 1) for line in dates.splitlines())
    not_before = parse_openssl_date(fields["notBefore"])
    not_after = parse_openssl_date(fields["notAfter"])
    assert (not_after - not_before).days == 365


def test_certificate_matches_private_key() -> None:
    key_modulus = run(["openssl", "rsa", "-noout", "-modulus", "-in", str(KEY)])
    crt_modulus = run(["openssl", "x509", "-noout", "-modulus", "-in", str(CRT)])
    assert key_modulus.replace("Modulus=", "") == crt_modulus.replace("Modulus=", "")


def test_pem_contains_key_and_certificate() -> None:
    pem_text = PEM.read_text(encoding="utf-8")
    assert "-----BEGIN PRIVATE KEY-----" in pem_text or "-----BEGIN RSA PRIVATE KEY-----" in pem_text
    assert "-----BEGIN CERTIFICATE-----" in pem_text
    assert "-----END CERTIFICATE-----" in pem_text


def test_verification_file_contains_required_details() -> None:
    text = VERIFICATION.read_text(encoding="utf-8")
    assert "subject=" in text
    assert f"O = {EXPECTED_ORG}" in text
    assert f"CN = {EXPECTED_CN}" in text
    assert "notBefore=" in text and "notAfter=" in text
    assert re.search(r"sha256 Fingerprint=([0-9A-F]{2}:){31}[0-9A-F]{2}", text), text


def test_check_cert_script_loads_certificate_and_reports_success() -> None:
    output = run(["python3", str(CHECK_SCRIPT)])
    assert f"Common Name: {EXPECTED_CN}" in output
    assert re.search(r"Expiration Date: \d{4}-\d{2}-\d{2}", output), output
    assert "Certificate verification successful" in output


def main() -> int:
    tests = [
        test_required_files_exist,
        test_private_key_is_2048_bit_rsa_and_mode_600,
        test_certificate_subject_and_validity,
        test_certificate_matches_private_key,
        test_pem_contains_key_and_certificate,
        test_verification_file_contains_required_details,
        test_check_cert_script_loads_certificate_and_reports_success,
    ]
    for test in tests:
        test()
        print(f"PASS {test.__name__}")
    print("All certificate requirement checks passed")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
