#!/usr/bin/env python3
"""Count basic feasible corner points for LP instances A_i x = b_i, x >= 0.

For each matching /app/A_i.csv and /app/b_i.csv pair, every 6-column basis is
examined.  The number of bases whose basic solution is componentwise
nonnegative is written to /app/corner_counts.txt, one integer per line.
"""

from __future__ import annotations

import glob
import itertools
import os
import re
from typing import Iterable, Iterator

import numpy as np

APP_DIR = "/app"
N_ROWS = 6
N_COLS = 40
BASIS_SIZE = 6
BATCH_SIZE = 100_000
TOL = 1e-9


def instance_indices() -> list[int]:
    """Return sorted indices i for which both A_i.csv and b_i.csv exist."""
    a_indices: set[int] = set()
    b_indices: set[int] = set()
    a_re = re.compile(r"A_(\d+)\.csv$")
    b_re = re.compile(r"b_(\d+)\.csv$")

    for path in glob.glob(os.path.join(APP_DIR, "A_*.csv")):
        match = a_re.search(os.path.basename(path))
        if match:
            a_indices.add(int(match.group(1)))
    for path in glob.glob(os.path.join(APP_DIR, "b_*.csv")):
        match = b_re.search(os.path.basename(path))
        if match:
            b_indices.add(int(match.group(1)))

    if a_indices != b_indices:
        missing_b = sorted(a_indices - b_indices)
        missing_a = sorted(b_indices - a_indices)
        raise FileNotFoundError(
            f"Mismatched A/b files: missing b for {missing_b}, missing A for {missing_a}"
        )
    return sorted(a_indices)


def batched(iterable: Iterable[tuple[int, ...]], batch_size: int) -> Iterator[np.ndarray]:
    """Yield combinations from iterable as int arrays of shape (<=batch_size, 6)."""
    iterator = iter(iterable)
    while True:
        chunk = list(itertools.islice(iterator, batch_size))
        if not chunk:
            return
        yield np.asarray(chunk, dtype=np.intp)


def count_corners_for_instance(A: np.ndarray, b: np.ndarray) -> int:
    """Count all 6-variable basic feasible solutions for one instance."""
    A = np.asarray(A, dtype=float)
    b = np.asarray(b, dtype=float).reshape(-1)
    if A.shape != (N_ROWS, N_COLS):
        raise ValueError(f"Expected A shape {(N_ROWS, N_COLS)}, got {A.shape}")
    if b.shape != (N_ROWS,):
        raise ValueError(f"Expected b shape {(N_ROWS,)}, got {b.shape}")

    count = 0
    rhs = np.broadcast_to(b, (BATCH_SIZE, N_ROWS))
    for combos in batched(itertools.combinations(range(N_COLS), BASIS_SIZE), BATCH_SIZE):
        # A[:, combos] would create axes in an inconvenient order; transpose to
        # obtain a stack of square matrices with shape (batch, 6, 6).
        matrices = A[:, combos].transpose(2, 0, 1)
        try:
            solutions = np.linalg.solve(matrices, rhs[: len(combos)])
            feasible = np.all(solutions >= -TOL, axis=1)
            count += int(np.count_nonzero(feasible))
        except np.linalg.LinAlgError:
            # Random continuous matrices are nonsingular almost surely, but this
            # fallback handles any singular candidate bases safely.
            for combo in combos:
                matrix = A[:, combo]
                try:
                    solution = np.linalg.solve(matrix, b)
                except np.linalg.LinAlgError:
                    continue
                if np.all(solution >= -TOL):
                    count += 1
    return count


def main() -> None:
    indices = instance_indices()
    if not indices:
        raise FileNotFoundError("No /app/A_i.csv and /app/b_i.csv instances found")

    counts: list[int] = []
    for i in indices:
        A = np.loadtxt(os.path.join(APP_DIR, f"A_{i}.csv"), delimiter=",")
        b = np.loadtxt(os.path.join(APP_DIR, f"b_{i}.csv"), delimiter=",")
        count = count_corners_for_instance(A, b)
        counts.append(count)
        print(f"instance {i}: {count}", flush=True)

    output_path = os.path.join(APP_DIR, "corner_counts.txt")
    with open(output_path, "w", encoding="utf-8") as f:
        for count in counts:
            f.write(f"{count}\n")


if __name__ == "__main__":
    main()
