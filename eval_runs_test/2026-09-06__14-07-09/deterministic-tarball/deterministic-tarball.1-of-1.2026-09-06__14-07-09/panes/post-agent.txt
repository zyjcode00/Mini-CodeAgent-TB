excluded_dirs = {".git", "node_modules", "tmp", ".cache", "__pycache__"}
excluded_suffixes = (".tmp", ".pyc", ".o")
text_extensions = {
    ".txt", ".md", ".py", ".sh", ".c", ".h", ".js",
    ".css", ".html", ".xml", ".json", ".yaml", ".yml",
}

if not src_dir.is_dir():
    raise SystemExit(f"source directory not found: {src_dir}")


def is_excluded_path(rel: Path) -> bool:
    return any(part in excluded_dirs for part in rel.parts)


def is_text_file(path: Path) -> bool:
    suffix = path.suffix.lower()
    if suffix in text_extensions:
        return True
    try:
        mime = subprocess.check_output(
            ["file", "--mime-type", "-b", str(path)],
            text=True,
        ).strip()
    except Exception:
        return False
    return mime.startswith("text/")


def should_exclude_file(src_path: Path, rel: Path, st: os.stat_result) -> bool:
    if is_excluded_path(rel.parent):
        return True
    if rel.name.endswith(excluded_suffixes):
        return True
    if stat.S_ISREG(st.st_mode) and stat.S_IMODE(st.st_mode) == 0o600:
        return True
    return False


def normalize_text(data: bytes) -> bytes:
    return data.replace(b"\r\n", b"\n").replace(b"\r", b"\n")

    stage = Path(tmp) / "stage"
    stage.mkdir(mode=0o755)

    # First pass: create directory skeleton in sorted order.
    for root, dirs, files in os.walk(src_dir, topdown=True, followlinks=False):
        root_path = Path(root)
        rel_root = root_path.relative_to(src_dir)
        dirs[:] = sorted(d for d in dirs if d not in excluded_dirs)
        (stage / rel_root).mkdir(parents=True, exist_ok=True)

            os.chmod(stage / rel_root / d, 0o755)

    for root, dirs, files in os.walk(src_dir, topdown=True, followlinks=False):

import hashlib
import os
import shutil
import stat
import subprocess
import tarfile
from pathlib import Path

ROOT = Path('/app')
BUILD = ROOT / 'build.sh'
OUT = ROOT / 'release.tar.zst'


def run_build(epoch: int) -> bytes:
    env = os.environ.copy()
    env['SOURCE_DATE_EPOCH'] = str(epoch)
    subprocess.run([str(BUILD)], cwd=ROOT, env=env, check=True)
    return OUT.read_bytes()


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def extract_archive() -> list[str]:
    tmp = ROOT / '.test_extract'
    if tmp.exists():
        shutil.rmtree(tmp)
    tmp.mkdir()
    subprocess.run(['tar', '--use-compress-program=zstd -d', '-xf', str(OUT), '-C', str(tmp)], shell=False, check=True)
    return [str(p.relative_to(tmp)) for p in sorted(tmp.rglob('*'))]


def test_build_script_executable():
    assert BUILD.exists()
    assert os.access(BUILD, os.X_OK)


def test_reproducible_build_and_contents():
    epoch = 1700000000
    data1 = run_build(epoch)
    data2 = run_build(epoch)
    assert sha256(data1) == sha256(data2)

    # Ensure changing SOURCE_DATE_EPOCH changes the archive metadata in a controlled way.
    data3 = run_build(epoch + 1)
    assert sha256(data1) != sha256(data3)
    run_build(epoch)

    tmp = ROOT / '.test_extract'
    if tmp.exists():
        shutil.rmtree(tmp)
    tmp.mkdir()
    subprocess.run(['zstd', '-d', '--stdout', str(OUT)], check=True, stdout=subprocess.PIPE)

    with subprocess.Popen(['zstd', '-d', '--stdout', str(OUT)], stdout=subprocess.PIPE) as p:
        assert p.stdout is not None
        tar_bytes = p.stdout.read()
        p.wait()

    tar_path = ROOT / '.test_archive.tar'
    tar_path.write_bytes(tar_bytes)
    try:
        with tarfile.open(tar_path, 'r:') as tf:
            names = tf.getnames()
            assert '.git' not in ''.join(names)
            assert all('node_modules' not in n for n in names)
            assert all('/tmp' not in n and '.cache' not in n for n in names)
            assert all('__pycache__' not in n for n in names)
            assert all(not n.endswith(('.tmp', '.pyc', '.o')) for n in names)

            members = {m.name: m for m in tf.getmembers()}
            assert 'src/src/windows.py' in members
            assert 'src/src/readonly.txt' not in members
            assert 'src/src/main.o' not in members
            assert 'src/tmp/test.tmp' not in members
            assert 'src/keys/private.key' not in members
            assert 'src/secrets/db.conf' not in members
            assert 'src/src/passwd-link' in members

            for member in tf.getmembers():
                assert member.uid == 0 and member.gid == 0
                assert member.uname == '' and member.gname == ''
                assert member.mtime == epoch
                if member.isdir():
                    assert stat.S_IMODE(member.mode) == 0o755
                elif member.issym():
                    continue
                else:
                    assert stat.S_IMODE(member.mode) in (0o644, 0o755)

            for name in ['src/src/unix.py', 'src/src/windows.py', 'src/src/file with spaces.txt', 'src/docs/README.txt']:
                data = tf.extractfile(name).read()
                assert b'\r' not in data

            # Binary/mixed content should remain unchanged and not be forced to LF-only.
            mixed = tf.extractfile('src/src/mixed.dat').read()
            assert mixed == (ROOT / 'src/src/mixed.dat').read_bytes()

            # Verify symlink preserved as symlink.
            passwd = members['src/src/passwd-link']
            assert passwd.issym()
    finally:
        if tar_path.exists():
            tar_path.unlink()
        if tmp.exists():
            shutil.rmtree(tmp)

import os
import stat
import subprocess
import sys
import tempfile
from pathlib import Path

src_dir = Path(sys.argv[1]).resolve()
out_file = Path(sys.argv[2]).resolve()
epoch = int(os.environ.get("SOURCE_DATE_EPOCH", "946684800"))

excluded_dirs = {".git", "node_modules", "tmp", ".cache", "__pycache__"}
excluded_suffixes = (".tmp", ".pyc", ".o")
text_extensions = {
    ".txt", ".md", ".py", ".sh", ".c", ".h", ".js",
    ".css", ".html", ".xml", ".json", ".yaml", ".yml",
}

if not src_dir.is_dir():
    raise SystemExit(f"source directory not found: {src_dir}")


def is_excluded_path(rel: Path) -> bool:
    return any(part in excluded_dirs for part in rel.parts)


def is_text_file(path: Path) -> bool:
    suffix = path.suffix.lower()
    if suffix in text_extensions:
        return True
    try:
        mime = subprocess.check_output(
            ["file", "--mime-type", "-b", str(path)],
            text=True,
        ).strip()
    except Exception:
        return False
    return mime.startswith("text/")


def should_exclude_file(src_path: Path, rel: Path, st: os.stat_result) -> bool:
    if is_excluded_path(rel.parent):
        return True
    if rel.name.endswith(excluded_suffixes):
        return True
    if stat.S_ISREG(st.st_mode) and stat.S_IMODE(st.st_mode) == 0o600:
        return True
    return False


def normalize_text(data: bytes) -> bytes:
    return data.replace(b"\r\n", b"\n").replace(b"\r", b"\n")


with tempfile.TemporaryDirectory(prefix="deterministic-build-") as tmp:
    stage = Path(tmp) / "stage"
    stage.mkdir(mode=0o755)

    # First pass: create directory skeleton in sorted order.
    for root, dirs, files in os.walk(src_dir, topdown=True, followlinks=False):
        root_path = Path(root)
        rel_root = root_path.relative_to(src_dir)
        dirs[:] = sorted(d for d in dirs if d not in excluded_dirs)
        (stage / rel_root).mkdir(parents=True, exist_ok=True)
        os.chmod(stage / rel_root, 0o755)
        for d in dirs:
            (stage / rel_root / d).mkdir(parents=True, exist_ok=True)
            os.chmod(stage / rel_root / d, 0o755)

    # Second pass: populate files and symlinks.
    for root, dirs, files in os.walk(src_dir, topdown=True, followlinks=False):
        root_path = Path(root)
        rel_root = root_path.relative_to(src_dir)
        dirs[:] = sorted(d for d in dirs if d not in excluded_dirs)
        for name in sorted(files):
            src_path = root_path / name
            rel_path = src_path.relative_to(src_dir)
            st = os.lstat(src_path)

            if should_exclude_file(src_path, rel_path, st):
                continue

            dst_path = stage / rel_path
            dst_path.parent.mkdir(parents=True, exist_ok=True)

            if stat.S_ISLNK(st.st_mode):
                os.symlink(os.readlink(src_path), dst_path)
                continue

            if not stat.S_ISREG(st.st_mode):
                continue

            data = src_path.read_bytes()
            mode = 0o755 if (data.startswith(b"#!") or (st.st_mode & stat.S_IXUSR)) else 0o644
            if is_text_file(src_path):
                data = normalize_text(data)
            dst_path.write_bytes(data)
            os.chmod(dst_path, mode)

    # Normalize timestamps and directory modes.
    for path in sorted(stage.rglob("*"), key=lambda p: (len(p.parts), str(p))):
        try:
            os.utime(path, (epoch, epoch), follow_symlinks=False)
        except Exception:
            pass
        if path.is_dir():
            os.chmod(path, 0o755)
        elif path.is_file():
            current = path.stat().st_mode
            os.chmod(path, 0o755 if (current & stat.S_IXUSR) else 0o644)

    os.utime(stage, (epoch, epoch))
    os.chmod(stage, 0o755)

    if out_file.exists():
        out_file.unlink()

    tar_cmd = [
        "tar",
        "--format=gnu",
        "--sort=name",
        "--numeric-owner",
        "--owner=0",
        "--group=0",
        f"--mtime=@{epoch}",
        "-C",
        str(stage),
        "-cf",
        "-",
        ".",
    ]
    zstd_cmd = [
        "zstd",
        "-19",
        "--long",
        "--no-check",
        "--single-thread",
        "-c",
    ]

    env = os.environ.copy()
    env["LC_ALL"] = "C"
    env["LANG"] = "C"
    env["TZ"] = "UTC"

    with open(out_file, "wb") as fh:
        tar_proc = subprocess.Popen(tar_cmd, stdout=subprocess.PIPE, env=env)
        assert tar_proc.stdout is not None
        try:
            zstd_proc = subprocess.Popen(zstd_cmd, stdin=tar_proc.stdout, stdout=fh, env=env)
        finally:
            tar_proc.stdout.close()
        zstd_rc = zstd_proc.wait()
        tar_rc = tar_proc.wait()
        if tar_rc != 0:
            raise SystemExit(tar_rc)
        if zstd_rc != 0:
            raise SystemExit(zstd_rc)

import hashlib

def run_build(epoch: int) -> bytes:
    env = os.environ.copy()
    env['SOURCE_DATE_EPOCH'] = str(epoch)
    subprocess.run([str(BUILD)], cwd=ROOT, env=env, check=True)
    return OUT.read_bytes()


def sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def extract_archive() -> list[str]:
    tmp = ROOT / '.test_extract'
    if tmp.exists():
        shutil.rmtree(tmp)
    tmp.mkdir()
    subprocess.run(['tar', '--use-compress-program=zstd -d', '-xf', str(OUT), '-C', str(tmp)], shell=False, check=True)
    return [str(p.relative_to(tmp)) for p in sorted(tmp.rglob('*'))]

    assert BUILD.exists()

    epoch = 1700000000
    data1 = run_build(epoch)
    data2 = run_build(epoch)
    assert sha256(data1) == sha256(data2)

    # Ensure changing SOURCE_DATE_EPOCH changes the archive metadata in a controlled way.
    data3 = run_build(epoch + 1)
    assert sha256(data1) != sha256(data3)
    run_build(epoch)

    tmp = ROOT / '.test_extract'
    if tmp.exists():

    with subprocess.Popen(['zstd', '-d', '--stdout', str(OUT)], stdout=subprocess.PIPE) as p:
        assert p.stdout is not None
        tar_bytes = p.stdout.read()
        p.wait()

    tar_path = ROOT / '.test_archive.tar'
    tar_path.write_bytes(tar_bytes)
    try:
        with tarfile.open(tar_path, 'r:') as tf:
            names = tf.getnames()

            assert all('node_modules' not in n for n in names)

def is_text_file(path: Path) -> bool:
    suffix = path.suffix.lower()
    if suffix in text_extensions:
        return True
    try:
        mime = subprocess.check_output(
            ["file", "--mime-type", "-b", str(path)],

        return False
    return mime.startswith("text/")
